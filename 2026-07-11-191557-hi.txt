  export async function magentoFetch<T>(
    query: string,
    variables?: Record<string, unknown>,
    opts?: { locale?: string; token?: string; revalidate?: number }
  ): Promise<T> {
    const headers: Record<string,string> = { "Content-Type": "application/json"
  };
    if (opts?.locale) headers["Store"] = storeView(opts.locale);
    if (opts?.token)  headers["Authorization"] = `Bearer ${opts.token}`;

    for (let attempt = 0; attempt < 2; attempt++) {          // retry once
      const res = await fetch(APP_CONFIG.magento.graphqlUrl, {
        method: "POST",
        headers,
        body: JSON.stringify({ query, variables }),
        next: opts?.revalidate ? { revalidate: opts.revalidate } : undefined,
      });
      const json = await res.json().catch(() => null);
      if (res.ok && !json?.errors) return json.data as T;
      if (attempt === 0 && res.status >= 500) continue;
      throw new MagentoError(json?.errors?.[0]?.message ?? `HTTP 
  ${res.status}`);
    }
  }

    const [state, setState] = useState<{data?:T; loading:boolean;
  error?:string}>({ loading: !!url });

  ---
  Phase 5 — Security Hardening (Design)

  Threat model / goals

  1. XSS token theft — token in localStorage is readable by any injected script.
  → Move to an httpOnly cookie (JS can't read it).
  2. Token leakage in logs — GET /api/orders?token=… puts the token in
  URLs/access logs. → Eliminate.
  3. No edge routing/guarding — no middleware; / doesn't redirect to a locale;
  /account is a client-only guard. → Add middleware.ts.

  Target architecture

  Login:   client POST /api/account {op:login,email,pw}
           → route calls Magento generateCustomerToken
           → route sets httpOnly cookie  ct=<token>  (Secure, SameSite=Lax, 30d)
           → returns { ok:true }  (token NEVER sent to JS)

  Any authed call: browser auto-sends ct cookie
           → route reads cookies().get('ct') server-side
           → forwards Bearer to Magento
           (client no longer reads or passes the token)

  Logout:  POST /api/account {op:logout} → route clears ct cookie
  Session: on load, auth-context calls refresh() → /api/account {op:customer}
           (route reads cookie) → customer or null. isLoggedIn = !!customer.
